import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useRef,
  useState,
  type ReactNode,
} from "react";
import { Link } from "react-router-dom";
import { useNotifications, type BellAlertItem } from "./NotificationContext";
import { recordAlertAction } from "../lib/alertActions";
import {
  SEVERITY_TONE,
  SENSOR_LABEL,
  type AirSeverity,
  type SensorType,
} from "../lib/notifications";
import { Badge } from "../ui";
import { Volume2, VolumeX } from "lucide-react";

/**
 * Alarm snooze length. One constant drives the toast button, its tooltip, the
 * notification dropdown's "Mute all" and the "Muted - rings again in m:ss"
 * label â€” change this line and every one of them follows.
 */
export const MUTE_SNOOZE_MS = 5 * 60 * 1000;
export const MUTE_SNOOZE_MINUTES = Math.round(MUTE_SNOOZE_MS / 60_000);

interface AlarmContextValue {
  unlocked: boolean;
  unlock: () => void;
  blocked: boolean;
  /** True while the dropdown's "Mute all" snooze is still counting down. */
  muteAllActive: boolean;
  muteAll: () => void;
  unmuteAll: () => void;
}

const AlarmContext = createContext<AlarmContextValue | null>(null);

export function useAlarm() {
  const ctx = useContext(AlarmContext);
  if (!ctx) throw new Error("useAlarm must be used within AlarmProvider");
  return ctx;
}

const UNLOCK_KEY = "air-alarm-unlocked";
const SNOOZE_KEY = "air-alarm-snooze";

/** Per-alert snooze end times: notification id -> epoch ms. */
type SnoozeMap = Record<string, number>;

interface SnoozeState {
  per: SnoozeMap;
  /** Blanket "mute all" end time (epoch ms); 0 when inactive. */
  all: number;
}

const EMPTY_SNOOZE: SnoozeState = { per: {}, all: 0 };

/**
 * Read snoozes back after a refresh so a reload does not restart a running
 * countdown. Expired entries are dropped here, which also stops the 1s
 * countdown ticker from starting for mutes that are already over.
 */
function loadSnooze(): SnoozeState {
  try {
    const raw = sessionStorage.getItem(SNOOZE_KEY);
    if (!raw) return EMPTY_SNOOZE;
    const parsed = JSON.parse(raw) as Partial<SnoozeState> | null;
    const now = Date.now();
    const per: SnoozeMap = {};
    if (parsed && typeof parsed.per === "object" && parsed.per) {
      for (const [id, until] of Object.entries(parsed.per)) {
        if (typeof until === "number" && until > now) per[id] = until;
      }
    }
    const all = typeof parsed?.all === "number" && parsed.all > now ? parsed.all : 0;
    return { per, all };
  } catch {
    // Private mode / disabled storage: snoozes then last for this render only.
    return EMPTY_SNOOZE;
  }
}

function saveSnooze(state: SnoozeState): void {
  try {
    sessionStorage.setItem(SNOOZE_KEY, JSON.stringify(state));
  } catch {
    // Ignore: quota or private mode must never break the alarm UI.
  }
}

/** Epoch ms when this alert's sound resumes, or 0 while it is audible. */
function alertMuteUntil(
  alert: Pick<BellAlertItem, "id" | "created_at">,
  state: SnoozeState,
  now: number
): number {
  const perUntil = state.per[String(alert.id)];
  let until = perUntil != null && perUntil > now ? perUntil : 0;
  if (state.all > now) {
    // "Mute all" silences the alerts that were already open when it was set.
    // An alert raised afterwards is NEW and rings immediately â€” the same rule
    // a per-alert snooze follows. Escalations ring too, because air-alerts
    // resolves the old row and inserts a new one when a level gets worse.
    const startedAt = state.all - MUTE_SNOOZE_MS;
    const raisedAt = new Date(alert.created_at).getTime();
    if (Number.isFinite(raisedAt) && raisedAt < startedAt) {
      until = Math.max(until, state.all);
    }
  }
  return until;
}

/** "4:32" for the muted-countdown label. */
function formatCountdown(ms: number): string {
  const totalSeconds = Math.max(0, Math.ceil(ms / 1000));
  const minutes = Math.floor(totalSeconds / 60);
  const seconds = totalSeconds % 60;
  return `${minutes}:${String(seconds).padStart(2, "0")}`;
}

/**
 * Plain-English rewrite of the stored alert text for the live toast:
 *   "A.I.R. ALERT [DRY] Hallway 2nd Floor: Humidity 0 %RH."
 * becomes
 *   "Humidity is 0 %RH at Hallway 2nd Floor."
 * Alerts without a numeric reading (offline) keep their stored wording â€”
 * there is no reading to phrase. The severity moves to the badge in the
 * toast header instead of the "[DRY]" prefix. Nothing here touches the
 * stored message, so SMS and alert creation are unchanged.
 */
function toastMessage(alert: BellAlertItem): string {
  if (alert.reading_value == null) return alert.message;
  const label = SENSOR_LABEL[alert.sensor_type as SensorType] ?? alert.sensor_type;
  const value =
    alert.sensor_type === "temp"
      ? alert.reading_value.toFixed(1)
      : String(Math.round(alert.reading_value));
  const reading = `${value} ${alert.reading_unit}`.trim();
  const location = `${alert.room_name} ${alert.floor}`.trim();
  return location
    ? `${label} is ${reading} at ${location}.`
    : `${label} is ${reading}.`;
}

type AlarmMode = "danger" | "warning" | null;

export function AlarmProvider({ children }: { children: ReactNode }) {
  const { alerts } = useNotifications();
  const [unlocked, setUnlocked] = useState<boolean>(() => {
    try { return sessionStorage.getItem(UNLOCK_KEY) === "1"; } catch { return false; }
  });
  const [blocked, setBlocked] = useState(false);
  const [snooze, setSnooze] = useState<SnoozeState>(() => loadSnooze());
  const [now, setNow] = useState<number>(() => Date.now());

  const audioCtxRef = useRef<AudioContext | null>(null);
  const nodesRef = useRef<{ osc?: OscillatorNode; gain?: GainNode; interval?: number }>({});

  const openAlerts = useMemo(
    () => alerts.filter((a) => a.status === "open"),
    [alerts]
  );
  const topAlert = openAlerts[0] ?? null;

  // Audible = open and not snoozed. A muted alert is simply left out, so a
  // new alert, or one whose level just got worse, still sounds even while a
  // different alert is muted. The toast, badge and bell are unaffected â€”
  // they read openAlerts, which never gets filtered.
  const audibleAlerts = useMemo(
    () => openAlerts.filter((a) => alertMuteUntil(a, snooze, now) <= now),
    [openAlerts, snooze, now]
  );
  const isDanger = audibleAlerts.some((a) => a.severity === "danger" || a.severity === "critical");
  const isWarning = !isDanger && audibleAlerts.some((a) => a.severity === "warning" || a.severity === "high");

  // Tick once a second while anything is snoozed: this drives the
  // "rings again in m:ss" countdown and brings the sound back the moment a
  // snooze expires (the filter above flips on the first tick past the end).
  const snoozeCountingDown =
    Object.values(snooze.per).some((until) => until > now) || snooze.all > now;
  useEffect(() => {
    if (!snoozeCountingDown) return;
    const timer = window.setInterval(() => setNow(Date.now()), 1000);
    return () => window.clearInterval(timer);
  }, [snoozeCountingDown]);

  // Persist every change so a page refresh does not restart a snooze.
  useEffect(() => { saveSnooze(snooze); }, [snooze]);

  const stopSound = useCallback(() => {
    const n = nodesRef.current;
    if (n.interval) window.clearInterval(n.interval);
    try { n.osc?.stop(); } catch { /* ignore */ }
    try { audioCtxRef.current?.close(); } catch { /* ignore */ }
    audioCtxRef.current = null;
    nodesRef.current = {};
  }, []);

  const startDanger = useCallback(() => {
    if (!audioCtxRef.current) {
      const ctx = new AudioContext();
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.type = "sawtooth";
      gain.gain.value = 0.15;
      osc.connect(gain).connect(ctx.destination);
      osc.start();
      audioCtxRef.current = ctx;
      nodesRef.current.osc = osc;
      nodesRef.current.gain = gain;
      let rising = true;
      let freq = 500;
      nodesRef.current.interval = window.setInterval(() => {
        freq += rising ? 40 : -40;
        if (freq >= 900) rising = false;
        if (freq <= 500) rising = true;
        osc.frequency.setValueAtTime(freq, ctx.currentTime);
      }, 50);
    }
    audioCtxRef.current?.resume().catch(() => setBlocked(true));
  }, []);

  const startWarning = useCallback(() => {
    if (!audioCtxRef.current) {
      const ctx = new AudioContext();
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.type = "square";
      gain.gain.value = 0.1;
      osc.frequency.value = 880;
      osc.connect(gain).connect(ctx.destination);
      osc.start();
      gain.gain.setValueAtTime(0, ctx.currentTime);
      audioCtxRef.current = ctx;
      nodesRef.current.osc = osc;
      nodesRef.current.gain = gain;
      // staggered beeping: 200ms on, 800ms off
      let on = true;
      nodesRef.current.interval = window.setInterval(() => {
        const t = ctx.currentTime;
        if (on) {
          gain.gain.cancelScheduledValues(t);
          gain.gain.setValueAtTime(0.1, t);
        } else {
          gain.gain.cancelScheduledValues(t);
          gain.gain.setValueAtTime(0, t);
        }
        on = !on;
      }, 200);
    }
    audioCtxRef.current?.resume().catch(() => setBlocked(true));
  }, []);

  // null = silent: sound locked, nothing open, or everything ringing is
  // snoozed. Sound still only ever starts after unlock(), so the browser's
  // autoplay rules are respected exactly as before.
  const mode: AlarmMode = !unlocked
    ? null
    : isDanger
    ? "danger"
    : isWarning
    ? "warning"
    : null;
  const modeRef = useRef<AlarmMode>(null);

  useEffect(() => {
    if (mode === null) {
      stopSound();
      modeRef.current = null;
      return;
    }
    // Restart when the tone itself changes (a warning escalating to a danger
    // wail) â€” otherwise the previous oscillator keeps playing its old tone.
    if (modeRef.current !== mode) {
      stopSound();
      modeRef.current = mode;
    }
    if (mode === "danger") startDanger();
    else startWarning();
  }, [mode, startDanger, startWarning, stopSound]);

  useEffect(() => () => stopSound(), [stopSound]);

  const unlock = useCallback(() => {
    try {
      const ctx = new (window as any).AudioContext();
      ctx.resume().then(() => {
        ctx.close();
        setUnlocked(true);
        sessionStorage.setItem(UNLOCK_KEY, "1");
        setBlocked(false);
      }).catch(() => setBlocked(true));
    } catch {
      setBlocked(true);
    }
  }, []);

  const handleAck = useCallback(async () => {
    if (!topAlert) return;
    await recordAlertAction({ notificationId: topAlert.id }, "acknowledge");
  }, [topAlert]);

  /** Snooze ONE alert. Everything else keeps ringing as it does today. */
  const muteAlert = useCallback((alert: BellAlertItem) => {
    const at = Date.now();
    setNow(at);
    setSnooze((prev) => ({
      ...prev,
      per: { ...prev.per, [String(alert.id)]: at + MUTE_SNOOZE_MS },
    }));
  }, []);

  const unmuteAlert = useCallback((alert: BellAlertItem) => {
    const at = Date.now();
    setNow(at);
    setSnooze((prev) => {
      const per = { ...prev.per };
      delete per[String(alert.id)];
      // Clear a blanket mute too: if "mute all" is what is holding this
      // alert down, leaving it set would make the click do nothing visible.
      return { per, all: 0 };
    });
  }, []);

  const muteAll = useCallback(() => {
    const at = Date.now();
    setNow(at);
    setSnooze((prev) => ({ ...prev, all: at + MUTE_SNOOZE_MS }));
  }, []);

  const unmuteAll = useCallback(() => {
    const at = Date.now();
    setNow(at);
    setSnooze((prev) => ({ ...prev, all: 0 }));
  }, []);

  const muteAllActive = snooze.all > now;

  const value = useMemo<AlarmContextValue>(
    () => ({ unlocked, unlock, blocked, muteAllActive, muteAll, unmuteAll }),
    [unlocked, unlock, blocked, muteAllActive, muteAll, unmuteAll]
  );

  const topMuteUntil = topAlert ? alertMuteUntil(topAlert, snooze, now) : 0;
  const topMuted = topMuteUntil > now;

  return (
    <AlarmContext.Provider value={value}>
      {children}
      {/* Enable sound prompt */}
      {!unlocked && (
        <div className="fixed bottom-4 right-4 z-50 bg-white border border-slate-200 rounded-xl shadow-lg p-4 max-w-sm">
          <p className="text-sm font-medium text-slate-800 mb-2">Enable alarm sound</p>
          <p className="text-xs text-slate-500 mb-3">Browsers block autoplay. Click to unlock the alarm tone for this session.</p>
          <button onClick={unlock} className="text-xs px-3 py-1.5 rounded-lg bg-teal-600 text-white">Enable</button>
          {blocked && <p className="text-xs text-red-600 mt-2">Audio is blocked. Check browser settings.</p>}
        </div>
      )}
      {blocked && unlocked && (
        <div className="fixed bottom-4 right-4 z-50 bg-red-50 border border-red-200 rounded-xl p-3 text-xs text-red-700">
          Alarm sound blocked by browser.
        </div>
      )}
      {/* Floating alarm banner */}
      {topAlert && (
        <div className="fixed top-4 right-4 z-50 bg-white border border-red-200 rounded-xl shadow-lg p-3 max-w-sm">
          <div className="flex items-center gap-2">
            <Badge
              label={topAlert.severity.toUpperCase()}
              tone={SEVERITY_TONE[(topAlert.severity as AirSeverity) || "warning"]}
            />
            <p className="text-xs font-semibold text-slate-800 truncate">{topAlert.room_name}</p>
          </div>
          <p className="text-xs text-slate-600 mt-1">{toastMessage(topAlert)}</p>
          {topMuted && (
            <p className="text-[10px] text-slate-400 mt-1">
              Muted - rings again in {formatCountdown(topMuteUntil - now)}
            </p>
          )}
          <div className="flex items-center gap-2 mt-2">
            <button onClick={handleAck} className="text-xs px-2.5 py-1 rounded-lg bg-teal-600 text-white">Acknowledge</button>
            <Link to="/notifications" className="text-xs text-teal-600 underline">View alert</Link>
            <button
              onClick={() => { if (topMuted) unmuteAlert(topAlert); else muteAlert(topAlert); }}
              title={topMuted ? "Unmute" : `Mute for ${MUTE_SNOOZE_MINUTES} minutes`}
              aria-label={topMuted ? "Unmute" : `Mute for ${MUTE_SNOOZE_MINUTES} minutes`}
              aria-pressed={topMuted}
              className="ml-auto w-9 h-9 shrink-0 rounded-lg border border-slate-200 flex items-center justify-center text-slate-500 hover:bg-slate-50 hover:text-slate-700 focus:outline-none focus:ring-2 focus:ring-teal-500"
            >
              {topMuted
                ? <VolumeX className="w-4 h-4" aria-hidden="true" />
                : <Volume2 className="w-4 h-4" aria-hidden="true" />}
            </button>
          </div>
        </div>
      )}
    </AlarmContext.Provider>
  );
}
